        // حساب المعدل الفصلي والتراكمي معاً
        function calculateGPA() {
            const system = parseInt(document.getElementById('gpa-system').value);
            const hoursElements = document.querySelectorAll('.hours');
            const gradeElements = document.querySelectorAll('.grade');
            
            let currentSemesterPoints = 0, currentSemesterHours = 0;

            for (let i = 0; i < hoursElements.length; i++) {
                const hours = parseFloat(hoursElements[i].value);
                const gradeValue = parseFloat(gradeElements[i].value);
                
                if (hours < 0 || hours > 10) {
                    Swal.fire({ icon: 'error', title: 'خطأ في المدخلات', text: 'يرجى إدخال عدد ساعات منطقي للمواد بين 1 و 10 ساعات!' });
                    return;
                }

                if (!isNaN(hours) && hours > 0) {
                    currentSemesterHours += hours;
                    currentSemesterPoints += (hours * gradeValue);
                }
            }

            if (currentSemesterHours === 0) {
                Swal.fire({ icon: 'error', title: 'خطأ', text: 'يرجى إدخال عدد الساعات للمواد الحالية أولاً!' });
                return;
            }

            const semGPA = (currentSemesterPoints / currentSemesterHours).toFixed(2);
            let resultHTML = `المعدل الفصلي المتوقع: <span style="color:#10b981;">${semGPA} من ${system}</span>`;

            const oldGpaInput = parseFloat(document.getElementById('old-gpa').value);
            const oldHoursInput = parseFloat(document.getElementById('old-hours').value);

            if (!isNaN(oldGpaInput)) {
                if(oldGpaInput < 0 || oldGpaInput > system) {
                    Swal.fire({ icon: 'error', title: 'خطأ بالتحقق', text: `المعدل السابق لا يمكن أن يكون سالباً أو أكبر من نظام الجامعة (${system})!` });
                    return;
                }
            }

            if (!isNaN(oldGpaInput) && !isNaN(oldHoursInput) && oldHoursInput > 0) {
                const totalHoursCombined = oldHoursInput + currentSemesterHours;
                const totalPointsCombined = (oldGpaInput * oldHoursInput) + currentSemesterPoints;
                const cumulativeGPA = (totalPointsCombined / totalHoursCombined).toFixed(2);
                resultHTML += `<br>المعدل التراكمي الإجمالي الجديد: <span style="color:#4f46e5;">${cumulativeGPA} من ${system}</span>`;
            }

            const resBox = document.getElementById('gpa-result');
            resBox.innerHTML = resultHTML;
            resBox.style.display = 'block';
            Swal.fire({ icon: 'success', title: 'تم الحساب بنجاح!', text: 'تم تحديث نتائج المعدل بالأسفل.' });
        }

        function sendWebNotification(title, bodyText) {
            if ("Notification" in window && Notification.permission === "granted") {

        function playBeep() {
            const context = new (window.AudioContext || window.webkitAudioContext)();
            const osc = context.createOscillator();
            const gain = context.createGain();
            osc.connect(gain);
            gain.connect(context.destination);
            osc.type = "sine";
            osc.frequency.value = 880; 
            gain.gain.setValueAtTime(1, context.currentTime);
            gain.gain.exponentialRampToValueAtTime(0.001, context.currentTime + 1);
            osc.start(context.currentTime);
            osc.stop(context.currentTime + 1);
        }

        function initTimer() {
            const workMin = parseInt(document.getElementById('work-duration').value) || 25;
            const breakMin = parseInt(document.getElementById('break-duration').value) || 5;
            
            if (workMin <= 0 || breakMin <= 0) {
                Swal.fire({ icon: 'error', title: 'خطأ بالتحقق', text: 'دقائق المؤقت يجب أن تكون أكبر من الصفر!' });
                return;
            }
            timerSeconds = (isWorkSession ? workMin : breakMin) * 60;
            updateTimerDisplay();
        }

        function updateTimerDisplay() {
            const minutes = Math.floor(timerSeconds / 60);
            const remainderSeconds = timerSeconds % 60;
            document.getElementById('timer-display').innerText = `${minutes < 10 ? '0' : ''}${minutes}:${remainderSeconds < 10 ? '0' : ''}${remainderSeconds}`;
        }

        function startTimer() {
            if (isRunning) return;
            requestNotificationPermission();
            if (timerSeconds === undefined) initTimer();
            isRunning = true;
            document.getElementById('start-btn').disabled = true;

            expectedEndTime = Date.now() + (timerSeconds * 1000);

            backgroundInterval = setInterval(() => {
                const timeLeft = Math.max(0, Math.round((expectedEndTime - Date.now()) / 1000));
                timerSeconds = timeLeft;
                updateTimerDisplay();

                if (timerSeconds <= 0) {
                    clearInterval(backgroundInterval);
                    isRunning = false;
                    document.getElementById('start-btn').disabled = false;
                    playBeep(); 
                    
                    if (isWorkSession) {
                        isWorkSession = false;
                        // الميزة 4: زيادة ساعات بومودورو المسجلة محلياً
                        const currentWorkMin = parseInt(document.getElementById('work-duration').value) || 25;
                        let pomoMinutes = parseFloat(localStorage.getItem('gamify-pomo-minutes')) || 0;
                        pomoMinutes += currentWorkMin;
                        localStorage.setItem('gamify-pomo-minutes', pomoMinutes);
                        updateGamificationStats();

                        sendWebNotification("انتهى وقت التركيز!", "حان وقت الراحة والاسترخاء! ☕");
                        Swal.fire({ icon: 'info', title: 'وقت الراحة! ☕', text: 'أحسنت تركيزك! خذ الآن قسطاً من الراحة.' });
                        document.getElementById('timer-status').innerText = "الوضعية الحالية: وقت الراحة ☕";
                    } else {
                        isWorkSession = true;
                        sendWebNotification("انتهت الاستراحة!", "فلنستكمل العمل والمذاكرة بقوة! 📚");
                        Swal.fire({ icon: 'success', title: 'عد للمذاكرة 📚', text: 'انتهت الاستراحة، فلنستكمل العمل بقوة!' });
                        document.getElementById('timer-status').innerText = "الوضعية الحالية: وقت المذاكرة 📚";
                    }
                    initTimer();
                }
            }, 500);
        }

        function pauseTimer() {
            clearInterval(backgroundInterval);
            isRunning = false;
            document.getElementById('start-btn').disabled = false;
        }

        function resetTimer() {
            clearInterval(backgroundInterval);
            isRunning = false;

            isWorkSession = true;

        function calculateAbsence() {
            const name = document.getElementById('abs-subject-select').value;
            const allowedPct = parseFloat(document.getElementById('abs-allowed-pct').value) / 100;
            const weeklyHours = parseInt(document.getElementById('abs-weekly-hours').value);
            const currentAbs = parseInt(document.getElementById('abs-count').value);

            if(!name) {
                Swal.fire({ icon: 'warning', title: 'مادة غير محددة', text: 'يرجى اختيار المادة المراد احتساب غيابها!' });
                return;
            }

            if(isNaN(weeklyHours) || isNaN(currentAbs) || currentAbs < 0) {
                Swal.fire({ icon: 'error', title: 'خطأ', text: 'بيانات الساعات أو الغيابات غير صالحة الحساب!' });
                return;
            }

            const records = JSON.parse(localStorage.getItem('student-absence-records')) || {};
            records[name] = currentAbs;
            localStorage.setItem('student-absence-records', JSON.stringify(records));

            const totalHours = weeklyHours * 15; 
            const maxAllowedAbsenceHours = Math.floor(totalHours * allowedPct);
            const remaining = maxAllowedAbsenceHours - currentAbs;
            
            let alertColor = "#10b981"; 
            let statusText = "وضعك في السليم ومستقر تماماً. استمر!";
            
            if (remaining <= 0) {
                alertColor = "#ef4444"; 
                statusText = "خطر! لقد تجاوزت الحد المسموح، أنت معرض للحرمان الفوري (DN).";
            } else if (remaining <= 2) {
                alertColor = "#f59e0b"; 
                statusText = "انتبه! لقد دخلت منطقة الإنذار، متبقي لك القليل جداً على الحرمان.";
            }

            const pctUsed = Math.min((currentAbs / maxAllowedAbsenceHours) * 100, 100);

            const resDiv = document.getElementById('absence-result');
            resDiv.style.display = "block";
            resDiv.innerHTML = `
                <div>تحليل مادة: <strong>${name}</strong></div>
                <div>الحد الأقصى للغياب المسموح لك به طوال الترم: <span style="color:#ef4444">${maxAllowedAbsenceHours} محاضرة/ساعة</span></div>
                <div>عدد الغيابات المتبقية لك حتى خطر الحرمان: <span style="font-size:1.2rem; color:${alertColor}">${remaining >= 0 ? remaining : 0}</span></div>
                <div style="margin-top:10px;"><strong>حالة الطالب الحالية:</strong> ${statusText}</div>
                <div class="progress-bar-container">
                    <div class="progress-bar" style="width: ${pctUsed}%; background-color: ${alertColor};"></div>
                </div>
            `;
            
            renderAbsenceCharts(); 
        }

        function calculateFinalRequirement() {
            const currentScore = parseFloat(document.getElementById('current-score').value);
            const finalWeight = parseFloat(document.getElementById('total-exam-weight').value);
            const target = parseFloat(document.getElementById('target-grade').value);

            if (isNaN(currentScore) || isNaN(finalWeight) || currentScore < 0 || finalWeight <= 0) {
                Swal.fire({ icon: 'error', title: 'بيانات خاطئة', text: 'يرجى إدخال قيم موجبة وصحيحة لأعمال السنة ووزن النهائي!' });
                return;
            }

            if (currentScore > (100 - finalWeight)) {
                Swal.fire({ icon: 'error', title: 'تناقض بالدرجات', text: `أعمال السنة لا يمكن أن تتخطى المتبقي من وزن النهائي وهو (${100 - finalWeight}) درجات!` });
                return;
            }

            const neededScore = target - currentScore;
            const resDiv = document.getElementById('final-result');
            resDiv.style.display = "block";

            if (neededScore <= 0) {
                resDiv.innerHTML = `لقد ضمنت الهدف الذي تطمح له بالفعل! 🎉 لست بحاجة إلى أي درجة للوصول للهدف، فقط احضر الاختبار لتثبيته.`;
            } else if (neededScore > finalWeight) {
                resDiv.innerHTML = `للأسف الحسابات تخبرنا بأنه <span style="color:#ef4444">مستحيل رياضياً</span> الوصول لهذا الهدف لأنك تحتاج إلى (${neededScore.toFixed(1)}) وهي أعلى من درجة الفاينل الإجمالية (${finalWeight}). حاول ضبط هدف أخر.`;
            } else {
                resDiv.innerHTML = `لتحقيق هدفك، تحتاج إلى الحصول على <span style="color:#4f46e5; font-size:1.3rem;">${neededScore.toFixed(1)} من أصل ${finalWeight}</span> في ورقة الاختبار النهائي. شد حيلك! 🎯`;
            }
        }

        function renderSchedule() {
            const schedule = JSON.parse(localStorage.getItem('student-schedule')) || [];
            const tbody = document.getElementById('schedule-tbody');
            if(!tbody) return;
            tbody.innerHTML = '';

            const globalSubjects = JSON.parse(localStorage.getItem('global-subjects')) || [];
            const activeSchedule = schedule.filter(item => globalSubjects.some(s => s.name === item.subject));

            activeSchedule.sort((a,b) => a.time.localeCompare(b.time));

            activeSchedule.forEach((item, index) => {

        function checkNextClass() {
            const schedule = JSON.parse(localStorage.getItem('student-schedule')) || [];
            const globalSubjects = JSON.parse(localStorage.getItem('global-subjects')) || [];
            const activeSchedule = schedule.filter(item => globalSubjects.some(s => s.name === item.subject));

            if(activeSchedule.length === 0) {
                Swal.fire({ icon: 'info', title: 'الجدول فارغ', text: 'قم بإضافة بعض المحاضرات لموادك الحالية أولاً لتفعيل ميزة التتبع المتقدمة الحية.' });
                return;
            }

            const daysWeight = { 'الأحد': 0, 'الإثنين': 1, 'الثلاثاء': 2, 'الأربعاء': 3, 'الخميس': 4, 'الجمعة': 5, 'السبت': 6 };
            
            const now = new Date();
            let currentDayIdx = now.getDay(); 

            const currentTimeString = `${now.getHours().toString().padStart(2, '0')}:${now.getMinutes().toString().padStart(2, '0')}`;

            let absoluteNextClass = null;
            let minimumDiffMinutes = Infinity;

            activeSchedule.forEach(item => {
                const classDayIdx = daysWeight[item.day];
                
                let dayDiff = classDayIdx - currentDayIdx;
                if (dayDiff < 0) {
                    dayDiff += 7; 
                }

                const [cHours, cMinutes] = currentTimeString.split(':').map(Number);
                const [classHours, classMinutes] = item.time.split(':').map(Number);

                let totalClassMinutes = (dayDiff * 24 * 60) + (classHours * 60 + classMinutes);
                let totalNowMinutes = (cHours * 60 + cMinutes);

                let diffMinutes = totalClassMinutes - totalNowMinutes;

                if (dayDiff === 0 && item.time <= currentTimeString) {
                    diffMinutes += 7 * 24 * 60;
                }

                if (diffMinutes < minimumDiffMinutes) {
                    minimumDiffMinutes = diffMinutes;
                    absoluteNextClass = item;
                }
            });

            document.querySelectorAll('#schedule-tbody tr').forEach(r => r.classList.remove('next-class-highlight'));

            if (absoluteNextClass) {
                const sortedSchedule = activeSchedule.sort((a,b) => a.time.localeCompare(b.time));
                const targetIdx = sortedSchedule.findIndex(i => i.subject === absoluteNextClass.subject && i.time === absoluteNextClass.time && i.day === absoluteNextClass.day);
                if(targetIdx !== -1) {
                    const rowEl = document.getElementById(`sched-row-${targetIdx}`);
                    if(rowEl) rowEl.classList.add('next-class-highlight');
                }

                const remainingDays = Math.floor(minimumDiffMinutes / (24 * 60));
                const remainingHours = Math.floor((minimumDiffMinutes % (24 * 60)) / 60);
                const remainingMins = minimumDiffMinutes % 60;

                let countdownText = "محاضرتك القادمة ";
                if(remainingDays > 0) countdownText += `بعد ${remainingDays} يوم و `;
                countdownText += `${remainingHours} ساعة و ${remainingMins} دقيقة.`;

                Swal.fire({
                    icon: 'success',
                    title: `المحاضرة القادمة: ${absoluteNextClass.subject}`,
                    html: `
                        <div style="text-align:right; font-size:0.95rem; line-height:1.6;">
                            <strong>اليوم:</strong> ${absoluteNextClass.day}<br>
                            <strong>الساعة:</strong> ${absoluteNextClass.time}<br>
                            <strong>القاعة:</strong> ${absoluteNextClass.room || 'غير محددة'}<br>
                            <strong>الدكتور:</strong> ${absoluteNextClass.doctor || '-'}<br>
                            <hr style="margin:10px 0;">
                            <span style="color:#eab308; font-weight:bold;"><i class="fa-solid fa-hourglass-half"></i> ${countdownText}</span>
                        </div>`
                });
            }
        }

        function exportToGoogleCalendar() {
            const schedule = JSON.parse(localStorage.getItem('student-schedule')) || [];
            if(schedule.length === 0) {
                Swal.fire({ icon: 'warning', title: 'الجدول فارغ', text: 'لا يوجد أي محاضرات لتصديرها!' });
                return;
            }

            const dayToIcalMap = { 'الأحد': 'SU', 'الإثنين': 'MO', 'الثلاثاء': 'TU', 'الأربعاء': 'WE', 'الخميس': 'TH' };
            
            let icsContent = "BEGIN:VCALENDAR\nVERSION:2.0\nPRODID:-//University Platform//Student Tool//EN\nCALSCALE:GREGORIAN\n";

            schedule.forEach(item => {
                const cleanTime = item.time.replace(':', '') + "00";
                const ruleDay = dayToIcalMap[item.day];
                
                if(!ruleDay) return;

                icsContent += "BEGIN:VEVENT\n";
                icsContent += `SUMMARY:${item.subject}\n`;
                icsContent += `DESCRIPTION:دكتور المادة: ${item.doctor || 'غير محدد'}\n`;
                icsContent += `LOCATION:قاعة رقم: ${item.room || 'غير محددة'}\n`;
                icsContent += `RRULE:FREQ=WEEKLY;BYDAY=${ruleDay}\n`;
                icsContent += `DTSTART;TZID=Asia/Riyadh:20260101T${cleanTime}\n`;
                icsContent += `DTEND;TZID=Asia/Riyadh:20260101T${cleanTime}\n`;
                icsContent += "END:VEVENT\n";
            });

            icsContent += "END:VCALENDAR";

            const blob = new Blob([icsContent], { type: 'text/calendar;charset=utf-8' });
            const link = document.createElement('a');
            link.href = URL.createObjectURL(blob);
            link.download = "university_schedule.ics";
            document.body.appendChild(link);
            link.click();
            document.body.removeChild(link);

            Swal.fire({
                icon: 'success',
                title: 'جاهز للربط الفوري!',
                text: 'تم تحميل ملف university_schedule.ics بنجاح. يمكنك استيراده تقويمك الشخصي.'
            });
        }

        // --- دوال رصد وتتبع الدرجات والأهداف ---
        function getLetterGrade(total) {
            if (total >= 95) return { letter: "امتياز مرتفع (+A)", color: "#10b981" };
            if (total >= 90) return { letter: "امتياز (A)", color: "#10b981" };
            if (total >= 85) return { letter: "جيد جداً مرتفع (+B)", color: "#3b82f6" };
            if (total >= 80) return { letter: "جيد جداً (B)", color: "#3b82f6" };
            if (total >= 75) return { letter: "جيد مرتفع (+C)", color: "#f59e0b" };
            if (total >= 70) return { letter: "جيد (C)", color: "#f59e0b" };
            if (total >= 60) return { letter: "مقبول (D)", color: "#718096" };
            return { letter: "راسب (F)", color: "#ef4444" };
        }

        function addTrackedGrade() {
            const subject = document.getElementById('track-subject-select').value;
            const quizzes = parseFloat(document.getElementById('track-quizzes').value) || 0;
            const mid = parseFloat(document.getElementById('track-mid').value) || 0;
            const project = parseFloat(document.getElementById('track-project').value) || 0;
            const final = parseFloat(document.getElementById('track-final').value) || 0;

            if(!subject) {
                Swal.fire({ icon: 'warning', title: 'بيانات ناقصة', text: 'يرجى اختيار المادة المستهدفة أولاً لرصد درجاتها.' });
                return;
            }

            if(quizzes < 0 || quizzes > 20 || mid < 0 || mid > 20 || project < 0 || project > 20 || final < 0 || final > 40) {
                Swal.fire({ icon: 'error', title: 'درجات غير منطقية', text: 'تأكد من أن الكويزات/الميد/المشروع من 20، والنهائي كحد أقصى 40 درجة!' });
                return;
            }

            const totalScore = quizzes + mid + project + final;
            const gradeData = { subject, quizzes, mid, project, final, totalScore };

            const grades = JSON.parse(localStorage.getItem('student-tracked-grades')) || [];
            
            const existingIdx = grades.findIndex(g => g.subject === subject);
            if(existingIdx !== -1) {
                grades[existingIdx] = gradeData;
            } else {
                grades.push(gradeData);
            }

            localStorage.setItem('student-tracked-grades', JSON.stringify(grades));

            renderTrackedGrades();
            renderGradesCharts(); 

            document.getElementById('track-quizzes').value = '';
            document.getElementById('track-mid').value = '';
            document.getElementById('track-project').value = '';
            document.getElementById('track-final').value = '';
        }

        // --- ميزة 1: مفكرة الاختبارات والواجبات الذكية (Deadlines) ---
        function addDeadlineItem() {
            const subject = document.getElementById('deadline-subject-select').value;
            const type = document.getElementById('deadline-type').value;
            const dateStr = document.getElementById('deadline-date').value;

            if(!subject || !dateStr) {
                Swal.fire({ icon: 'warning', title: 'بيانات غير مكتملة', text: 'يرجى اختيار المادة وتحديد تاريخ التسليم الحقيقي!' });
                return;
            }

            const deadlines = JSON.parse(localStorage.getItem('student-deadlines')) || [];
            deadlines.push({ subject, type, date: dateStr });
            localStorage.setItem('student-deadlines', JSON.stringify(deadlines));

            document.getElementById('deadline-date').value = '';
            renderDeadlines();
            checkAndAlertApproachingDeadlines();
        }

        function renderDeadlines() {
            const deadlines = JSON.parse(localStorage.getItem('student-deadlines')) || [];
            const globalSubjects = JSON.parse(localStorage.getItem('global-subjects')) || [];
            const tbody = document.getElementById('deadlines-tbody');
            if(!tbody) return;
            tbody.innerHTML = '';

            // تصفية المواد النشطة فقط
            const activeDeadlines = deadlines.filter(d => globalSubjects.some(s => s.name === d.subject));

            activeDeadlines.sort((a,b) => a.date.localeCompare(b.date));

            activeDeadlines.forEach((item, index) => {
                const today = new Date();
                today.setHours(0,0,0,0);
                const dDate = new Date(item.date);
                dDate.setHours(0,0,0,0);

                const timeDiff = dDate.getTime() - today.getTime();
                const daysLeft = Math.ceil(timeDiff / (1000 * 3600 * 24));

                let pillClass = "pill-green";
                let textRemaining = `متبقي ${daysLeft} يوم`;

                if (daysLeft < 0) {
                    pillClass = "pill-red";
                    textRemaining = "انتهى الموعد ⌛";
                } else if (daysLeft <= 2) {
                    pillClass = "pill-red";
                } else if (daysLeft <= 5) {
                    pillClass = "pill-yellow";
                }

                const tr = document.createElement('tr');
                tr.innerHTML = `
                    <td><strong>${item.subject}</strong></td>
                    <td><span class="btn-secondary" style="padding:2px 8px; border-radius:5px; font-size:0.85rem; background:#6366f1;">${item.type}</span></td>
                    <td>${item.date}</td>
                    <td><span class="countdown-pill ${pillClass}">${textRemaining}</span></td>
                    <td><button class="btn-danger" onclick="deleteDeadline(${index})" style="padding:4px 10px; font-size:0.8rem;"><i class="fa-solid fa-trash"></i></button></td>
                `;
                tbody.appendChild(tr);
            });
        }

        function deleteDeadline(index) {
            const deadlines = JSON.parse(localStorage.getItem('student-deadlines')) || [];
            deadlines.sort((a,b) => a.date.localeCompare(b.date));
            deadlines.splice(index, 1);
            localStorage.setItem('student-deadlines', JSON.stringify(deadlines));
            renderDeadlines();
        }

        function checkAndAlertApproachingDeadlines() {
            const deadlines = JSON.parse(localStorage.getItem('student-deadlines')) || [];
            const globalSubjects = JSON.parse(localStorage.getItem('global-subjects')) || [];
            const activeDeadlines = deadlines.filter(d => globalSubjects.some(s => s.name === d.subject));

            activeDeadlines.forEach(item => {
                const today = new Date();
                today.setHours(0,0,0,0);
                const dDate = new Date(item.date);
                dDate.setHours(0,0,0,0);

                const timeDiff = dDate.getTime() - today.getTime();
                const daysLeft = Math.ceil(timeDiff / (1000 * 3600 * 24));

                if(daysLeft >= 0 && daysLeft <= 2) {
                    sendWebNotification(`تنبيه اقتراب تسليم! ⚠️`, `المادة: ${item.subject} - ${item.type} متبقي عليه ${daysLeft} يوم فقط!`);
                }
            });
        }

        // --- ميزة 2: حاسبة المعدل الافتراضية وحساب وتوزيع السيناريو المطلوب لـ GPA ---
        function simulateGpaScenario() {
            const system = parseInt(document.getElementById('planner-gpa-system').value);
            const currentGpa = parseFloat(document.getElementById('planner-current-gpa').value);
            const passedHours = parseFloat(document.getElementById('planner-passed-hours').value);
            const targetGpa = parseFloat(document.getElementById('planner-target-gpa').value);

            const globalSubjects = JSON.parse(localStorage.getItem('global-subjects')) || [];
            
            if(globalSubjects.length === 0) {
                Swal.fire({ icon: 'warning', title: 'لا يوجد مواد', text: 'يرجى إضافة مواد الترم الحالي في القسم الأول لتحديد الخطة عليها!' });
                return;
            }

            if(isNaN(currentGpa) || isNaN(passedHours) || isNaN(targetGpa)) {
                Swal.fire({ icon: 'error', title: 'مدخلات ناقصة', text: 'يرجى تعبئة جميع الحقول الرياضية للمخطط الافتراضي!' });
                return;
            }

            if(targetGpa > system || currentGpa > system) {
                Swal.fire({ icon: 'error', title: 'خطأ منطقي', text: `لا يمكن للمعدل أن يتجاوز نظام النقاط المعتمد للجامعة وهو (${system})!` });
                return;
            }

            let currentTermHours = 0;
            globalSubjects.forEach(s => currentTermHours += s.hours);

            const totalHoursFuture = passedHours + currentTermHours;
            const targetTotalPoints = targetGpa * totalHoursFuture;
            const currentTotalPoints = currentGpa * passedHours;
            const pointsNeededInTerm = targetTotalPoints - currentTotalPoints;
            const targetTermGPA = pointsNeededInTerm / currentTermHours;

            const resDiv = document.getElementById('planner-result');
            resDiv.style.display = "block";

            if (targetTermGPA > system) {
                resDiv.innerHTML = `
                    <p style="color:#ef4444; font-weight:bold;"><i class="fa-solid fa-triangle-exclamation"></i> السيناريو مستحيل رياضياً!</p>
                    <p>الهدف يتطلب الحصول على معدل فصلي مقداره <strong>${targetTermGPA.toFixed(2)}</strong> في هذا الترم، وهو أعلى من سقف نظام الجامعة الحالية (${system}). حاول خفض الهدف قليلاً.</p>
                `;
                return;
            }

            let neededGradeText = "";
            if(system === 5) {
                if(targetTermGPA <= 2.0) neededGradeText = "مقبول (D) فما فوق";
                else if(targetTermGPA <= 3.0) neededGradeText = "جيد (C) في معظم المواد";
                else if(targetTermGPA <= 4.0) neededGradeText = "جيد جداً (B) في جميع موادك";
                else if(targetTermGPA <= 4.5) neededGradeText = "جيد جداً مرتفع (+B) إلى امتياز (A)";
                else neededGradeText = "امتياز مرتفع (+A) بالدرجات الكاملة دون تهاون!";
            } else {
                if(targetTermGPA <= 1.0) neededGradeText = "مقبول (D) فما فوق";
                else if(targetTermGPA <= 2.0) neededGradeText = "جيد (C) في أغلب المواد";
                else if(targetTermGPA <= 3.0) neededGradeText = "جيد جداً (B) كمتوسط عام";
                else neededGradeText = "امتياز (A) في كل المواد بلا استثناء!";
            }

            resDiv.innerHTML = `
                <h4 style="color:#06b6d4; margin-bottom:10px;"><i class="fa-solid fa-bullseye"></i> نتائج خطة محاكاة الهدف المفتوح:</h4>
                <p>إجمالي ساعاتك بعد هذا الترم ستصبح: <strong>${totalHoursFuture} ساعة</strong></p>
                <p>المعدل الفصلي المطلوب حده الأدنى في هذا الترم الحالي هو: <span style="font-size:1.3rem; color:#4f46e5;">${Math.max(0, targetTermGPA).toFixed(2)} من ${system}</span></p>
                <p><strong>توزيع التقديرات المقترح لتأمين الهدف لقائمة موادك الحالية:</strong> يلتزم الطالب بتحقيق معدل تقديري لا يقل عن <span style="color:#10b981; font-weight:bold;">[ ${neededGradeText} ]</span> في كل مادة من ملواده المسجلة الحالية والبالغ عددها (${globalSubjects.length} مواد).</p>
            `;

        }

        // --- ميزة 3: مساحة التدوين الذكي وحساب الكلمات وفلاش كاردز الملاحظات الذكية ---
        function countNoteWords() {
            const text = document.getElementById('note-textarea').value.trim();
            const count = text === "" ? 0 : text.split(/\s+/).length;
            document.getElementById('note-word-count').innerText = count;
        }

        function saveSmartNote() {
            const subject = document.getElementById('note-subject-select').value;
            const content = document.getElementById('note-textarea').value.trim();

            if(!subject || !content) {
                Swal.fire({ icon: 'warning', title: 'حقول ناقصة', text: 'يرجى اختيار مادة وكتابة محتوى الملاحظة لحفظها بالتطبيق!' });
                return;
            }

            const notes = JSON.parse(localStorage.getItem('student-smart-notes')) || [];
            notes.push({ subject, content, id: Date.now() });
            localStorage.setItem('student-smart-notes', JSON.stringify(notes));

            document.getElementById('note-textarea').value = '';
            document.getElementById('note-word-count').innerText = 0;
            renderSmartNotes();
            Swal.fire({ icon: 'success', title: 'تم الحفظ دفتر الملاحظات المدمج بنجاح!' });
        }

        function renderSmartNotes() {
            const notes = JSON.parse(localStorage.getItem('student-smart-notes')) || [];
            const globalSubjects = JSON.parse(localStorage.getItem('global-subjects')) || [];
            const container = document.getElementById('saved-notes-container');
            if(!container) return;
            container.innerHTML = '';

            const activeNotes = notes.filter(n => globalSubjects.some(s => s.name === n.subject));

            if(activeNotes.length === 0) {
                container.innerHTML = `<p style="font-size:0.9rem; color:var(--text-muted)">لا توجد ملاحظات محفوظة حتى الآن.</p>`;
                return;
            }

            activeNotes.forEach((item, index) => {
                const div = document.createElement('div');
                div.className = 'grade-tracker-item';
                div.style.background = 'var(--input-bg)';
                div.innerHTML = `
                    <div style="display:flex; justify-content:space-between; align-items:center; margin-bottom:8px;">
                        <span class="btn-secondary" style="padding:2px 8px; border-radius:5px; font-size:0.85rem;">${item.subject}</span>
                        <button class="btn-danger" onclick="deleteSmartNote(${item.id})" style="padding:4px 8px; font-size:0.75rem;"><i class="fa-solid fa-trash"></i></button>
                    </div>
                    <p style="white-space: pre-wrap; font-size:0.95rem; color:var(--text-main);">${item.content}</p>
                `;
                container.appendChild(div);
            });
        }

        function deleteSmartNote(id) {
            let notes = JSON.parse(localStorage.getItem('student-smart-notes')) || [];
            notes = notes.filter(n => n.id !== id);
            localStorage.setItem('student-smart-notes', JSON.stringify(notes));
            renderSmartNotes();
        }

        function generateFlashcardsFromNote() {
            const content = document.getElementById('note-textarea').value.trim();
            if(!content) {
                Swal.fire({ icon: 'warning', title: 'النص فارغ', text: 'اكتب نص الملاحظات أولاً لفرز الأسئلة منها وتوليد الفلاش كاردز تلقائياً!' });
                return;
            }

            // تقسيم النص لأسطر لاستخلاص الروابط أو المفاهيم
            const lines = content.split('\n');
            let generatedCount = 0;
            const flashcards = JSON.parse(localStorage.getItem('student-flashcards')) || [];

            lines.forEach(line => {
                let parts = [];
                if(line.includes(':')) {
                    parts = line.split(':');
                } else if(line.includes('=')) {
                    parts = line.split('=');
                } else if(line.includes('-')) {
                    parts = line.split('-');
                }

                if(parts.length >= 2) {
                    const question = parts[0].trim();
                    const answer = parts[1].trim();
                    if(question.length > 2 && answer.length > 2) {
                        flashcards.push({ question, answer });
                        generatedCount++;
                    }
                }
            });

            if(generatedCount > 0) {
                localStorage.setItem('student-flashcards', JSON.stringify(flashcards));
                renderFlashcards();
                Swal.fire({
                    icon: 'success',
                    title: `توليد ذكي ناجح! ⚡`,
                    text: `تم التعرف وتوليد (${generatedCount}) كارت استذكار ذكي وإضافتها لقسم الكروت تلقائياً دون تكرار تعبئتها يدويّاً!`
                });
            } else {
                Swal.fire({
                    icon: 'info',
                    title: 'لم نجد فواصل صالحة للتوليد',
                    text: 'لتوليد الكروت التلقائي، اكتب المفهوم ثم افصل بينه وبين التعريف باستخدام (نقطتين :) لتتحول تلقائياً لكروت متقاطعة.'
                });
            }
        }

        // --- ميزة 4: عداد التحفيز ولوحة إحصائيات عداد الإنجاز (Gamification Engine) ---
        function updateGamificationStats() {
            // 1. حساب ساعات بومودورو الكلية
            const pomoMinutes = parseFloat(localStorage.getItem('gamify-pomo-minutes')) || 0;
            const pomoHours = (pomoMinutes / 60).toFixed(1);
            const pomoEl = document.getElementById('gamify-pomo-hours');
            if(pomoEl) pomoEl.innerText = `${pomoHours} ساعة`;

            // 2. حساب نسبة إنجاز المهام العامة هذا الأسبوع
            const todos = JSON.parse(localStorage.getItem('university-todos')) || [];
            const pctEl = document.getElementById('gamify-todo-pct');
            const motivateEl = document.getElementById('gamify-motivate-text');
            
            if(todos.length === 0) {
                if(pctEl) pctEl.innerText = "0%";
                if(motivateEl) motivateEl.innerText = "لا يوجد مهام حالية! 🎯";
            } else {
                const completedCount = todos.filter(t => t.completed).length;
                const pct = Math.round((completedCount / todos.length) * 100);
                if(pctEl) pctEl.innerText = `${pct}%`;

                // تغيير رسائل التحفيز التفاعلية حسب النسبة
                if(pct === 100) {
                    if(motivateEl) motivateEl.innerText = "ملك الإنتاجية الأسبوعي! 🏆";
                } else if(pct >= 75) {
                    if(motivateEl) motivateEl.innerText = "أنجزت معظم مهامك، استمر! 🔥";
                } else if(pct >= 40) {
                    if(motivateEl) motivateEl.innerText = "تقدم رائع، تابع السعي! 📚";
                } else {
                    if(motivateEl) motivateEl.innerText = "خطوة بخطوة، ستحقق أهدافك! 💪";
                }
            }
        }

        // --- ميزة 5: منتدى مشاركة روابط الملفات وسحابة المادة (Peer Sharing Shortcuts) ---
        function addCloudLink() {
            const subject = document.getElementById('share-subject-select').value;
            const name = document.getElementById('share-resource-name').value.trim();
            const url = document.getElementById('share-resource-url').value.trim();

            if(!subject || !name || !url) {
                Swal.fire({ icon: 'warning', title: 'بيانات ناقصة', text: 'تأكد من اختيار المادة، كتابة اسم المجلد الدراسي، ولصق الرابط السحابي!' });
                return;
            }

            const cloudLinks = JSON.parse(localStorage.getItem('student-cloud-sharing')) || [];
            cloudLinks.push({ subject, name, url });
            localStorage.setItem('student-cloud-sharing', JSON.stringify(cloudLinks));

            document.getElementById('share-resource-name').value = '';
            document.getElementById('share-resource-url').value = '';
            renderCloudLinks();
            Swal.fire({ icon: 'success', title: 'تم تثبيت ملف المادة بنجاح بالسحابة الموحدة!' });
        }

        function renderCloudLinks() {
            const cloudLinks = JSON.parse(localStorage.getItem('student-cloud-sharing')) || [];
            const globalSubjects = JSON.parse(localStorage.getItem('global-subjects')) || [];
            const container = document.getElementById('peer-sharing-box');
            if(!container) return;
            container.innerHTML = '';

            const activeLinks = cloudLinks.filter(c => globalSubjects.some(s => s.name === c.subject));

            if(activeLinks.length === 0) {
                container.innerHTML = `<p style="grid-column: 1/-1; font-size:0.9rem; color:var(--text-muted)">لا توجد روابط تشاركية مرفوعة لموادك الحالية حتى الآن.</p>`;
                return;
            }

            activeLinks.forEach((item, index) => {
                const card = document.createElement('div');
                card.className = 'link-card';
                card.style.position = 'relative';
                card.innerHTML = `
                    <a href="${item.url}" target="_blank" style="text-decoration:none; color:inherit; display:flex; align-items:center; gap:12px; width:85%">
                        <i class="fa-solid fa-folder-open" style="color: #ec4899;"></i>
                        <div>
                            <strong>${item.name}</strong>
                            <p style="margin:0; font-size:0.8rem; color:var(--text-muted);">مادة: ${item.subject}</p>
                        </div>
                    </a>
                    <button class="btn-danger" onclick="deleteCloudLink(${index})" style="position:absolute; left:10px; top:15px; padding:4px 8px; font-size:0.75rem; border-radius:5px;">
                        <i class="fa-solid fa-trash"></i>
                    </button>
                `;
                container.appendChild(card);
            });
        }

        function deleteCloudLink(index) {
            const cloudLinks = JSON.parse(localStorage.getItem('student-cloud-sharing')) || [];
            cloudLinks.splice(index, 1);
            localStorage.setItem('student-cloud-sharing', JSON.stringify(cloudLinks));
            renderCloudLinks();
        }

        function importStudentData(fileInput) {
            const file = fileInput.files[0];
            if (!file) return;

            const reader = new FileReader();
            reader.onload = function(e) {
                try {
                    const parsedData = JSON.parse(e.target.result);
                    
                    if(parsedData.globalSubjects) localStorage.setItem('global-subjects', JSON.stringify(parsedData.globalSubjects));
                    if(parsedData.absenceRecords) localStorage.setItem('student-absence-records', JSON.stringify(parsedData.absenceRecords));
                    if(parsedData.todos) localStorage.setItem('university-todos', JSON.stringify(parsedData.todos));
                    if(parsedData.schedule) localStorage.setItem('student-schedule', JSON.stringify(parsedData.schedule));
                    if(parsedData.trackedGrades) localStorage.setItem('student-tracked-grades', JSON.stringify(parsedData.trackedGrades));
                    if(parsedData.flashcards) localStorage.setItem('student-flashcards', JSON.stringify(parsedData.flashcards));
                    if(parsedData.deadlines) localStorage.setItem('student-deadlines', JSON.stringify(parsedData.deadlines));
                    if(parsedData.smartNotes) localStorage.setItem('student-smart-notes', JSON.stringify(parsedData.smartNotes));
                    if(parsedData.cloudSharing) localStorage.setItem('student-cloud-sharing', JSON.stringify(parsedData.cloudSharing));
                    if(parsedData.pomoMinutes) localStorage.setItem('gamify-pomo-minutes', parsedData.pomoMinutes);
                    if(parsedData.darkMode) localStorage.setItem('student-dark-mode', parsedData.darkMode);
                    if(parsedData.themeName) localStorage.setItem('student-app-theme', parsedData.themeName);

                    Swal.fire({
                        icon: 'success', 
                        title: 'تم استعادة البيانات الشاملة!', 
                        text: 'يتم الآن تحديث المنصة...'
                    }).then(() => {
                        location.reload();
                    });

                } catch(err) {
                    Swal.fire({ icon: 'error', title: 'فشل الاستيراد', text: 'الملف المرفوع غير صالح أو تعرض للتلف.' });
                }
            };
            reader.readAsText(file);
        }
